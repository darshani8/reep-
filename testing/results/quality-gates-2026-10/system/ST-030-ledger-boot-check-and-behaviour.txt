import sys, requests, datetime as dt
B = sys.argv[1]; s = requests.Session()

today = dt.date.fromisoformat(s.get(f'{B}/api/student/ledger').json()['today'])
D = (today - dt.timedelta(days=2)).isoformat(); LOCKED = (today - dt.timedelta(days=3)).isoformat(); FUT = (today + dt.timedelta(days=1)).isoformat()
FULL = {'DAWN': {'SLEEPING': 4}, 'MORNING': {'LECTURES': 3}, 'MIDDAY': {'COURSEWORK': 3}, 'AFTERNOON': {'SKILLING': 3}, 'EVENING': {'LEISURE': 4}, 'NIGHT': {'SLEEPING': 7}}
def cells(m): return [{'slot': sl, 'activity': a, 'hours': h} for sl, d in m.items() for a, h in d.items()]
def out(label, r):
    try: j = r.json()
    except Exception: j = r.text
    det = j.get('detail') if isinstance(j, dict) and 'detail' in j else {k: j.get(k) for k in ('day', 'status', 'total_hours', 'can_submit', 'editable', 'locked', 'submit_blocked_reason')} if isinstance(j, dict) else j
    print(f'{label:58s} -> {r.status_code} {str(det)[:300]}')
print('day under test (today-2, inside the 2-day window):', 'today-2'); 
out('GET ledger?day=today-2 (empty)', s.get(f'{B}/api/student/ledger', params={'day': D}))
out('PUT cell over slot capacity (AFTERNOON SKILLING 3.5 h)', s.put(f'{B}/api/student/ledger', json={'day': D, 'cells': cells({'AFTERNOON': {'SKILLING': 3.5}})}))
out('PUT a slot summing past capacity (AFTERNOON 2+2 h)', s.put(f'{B}/api/student/ledger', json={'day': D, 'cells': cells({'AFTERNOON': {'SKILLING': 2, 'LECTURES': 2}})}))
out('PUT non-half-hour value (0.3 h)', s.put(f'{B}/api/student/ledger', json={'day': D, 'cells': cells({'DAWN': {'SLEEPING': 0.3}})}))
out('PUT 23 h (one hour short)', s.put(f'{B}/api/student/ledger', json={'day': D, 'cells': cells({**FULL, 'NIGHT': {'SLEEPING': 6}})}))
out('POST submit while 1 h short', s.post(f'{B}/api/student/ledger/submit', json={'day': D}))
out('PUT exactly 24 h', s.put(f'{B}/api/student/ledger', json={'day': D, 'cells': cells(FULL)}))
out('POST submit at 24 h', s.post(f'{B}/api/student/ledger/submit', json={'day': D}))
out('PUT after submit (latched)', s.put(f'{B}/api/student/ledger', json={'day': D, 'cells': cells(FULL)}))
out('POST copy-yesterday onto a submitted day', s.post(f'{B}/api/student/ledger/copy-yesterday', json={'day': D}))
out('PUT on a locked day (today-3)', s.put(f'{B}/api/student/ledger', json={'day': LOCKED, 'cells': cells(FULL)}))
out('PUT on tomorrow', s.put(f'{B}/api/student/ledger', json={'day': FUT, 'cells': cells(FULL)}))
out('GET ledger/history?days=5', s.get(f'{B}/api/student/ledger/history', params={'days': 5}))
