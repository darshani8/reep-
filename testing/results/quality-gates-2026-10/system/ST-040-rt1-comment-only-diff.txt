"""Prove a diff between two commits touches only comments: for every changed file, compare the
Python token streams with COMMENT/NL/NEWLINE-only tokens removed, and the AST dumps. Usage: comment_only.py OLD NEW PATH..."""
import ast, io, subprocess, sys, tokenize
old, new, paths = sys.argv[1], sys.argv[2], sys.argv[3:]
files = subprocess.run(['git', 'diff', '--name-only', old, new, '--', *paths], capture_output=True, text=True, check=True).stdout.split()
def show(rev, f):
    r = subprocess.run(['git', 'show', f'{rev}:{f}'], capture_output=True, text=True); return r.stdout if r.returncode == 0 else None
def toks(src):
    return [(t.type, t.string) for t in tokenize.generate_tokens(io.StringIO(src).readline) if t.type not in (tokenize.COMMENT, tokenize.NL)]
ok = True
print(f'changed files under {paths}: {len(files)}')
for f in files:
    a, b = show(old, f), show(new, f)
    if not f.endswith('.py') or a is None or b is None:
        print(f'NOT-PY-OR-ADDED/REMOVED {f}'); ok = False; continue
    ta, tb = toks(a), toks(b)
    same_tok = ta == tb
    same_ast = ast.dump(ast.parse(a)) == ast.dump(ast.parse(b))
    nl = subprocess.run(['git', 'diff', '--numstat', old, new, '--', f], capture_output=True, text=True).stdout.split()[:2]
    print(f"{'COMMENT-ONLY' if same_tok and same_ast else 'CODE-CHANGED'}  +{nl[0]}/-{nl[1]}  tokens_equal={same_tok} ast_equal={same_ast}  {f}")
    ok &= same_tok and same_ast
print('RESULT:', 'every change is comment-only' if ok else 'NON-COMMENT CHANGE FOUND'); sys.exit(0 if ok else 1)
