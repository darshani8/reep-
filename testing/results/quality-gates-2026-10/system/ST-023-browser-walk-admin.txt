// Browser walk, one screenshot per step. Usage: node walk.mjs ROLE OUTDIR CASEID
import { chromium } from '/home/user/reep-/node_modules/playwright/index.mjs';
const [role, out, cid] = process.argv.slice(2);
const BASE = 'http://localhost:4200';
const A = { student: ['student@bgscet.ac.in', 'student123', /^Student\b/], mentor: ['mentor@bgscet.ac.in', 'mentor123', /^Faculty\b/],
  alumni: ['alumni@bgscet.ac.in', 'alumni123', /^Alumni\b/], admin: ['admin@bgscet.ac.in', 'admin123', null] };
const ACCT = { student: /Test Student/, mentor: /Test Mentor/, alumni: /Alumn/i, admin: /Main Admin/ };
const SCREENS = { student: ['/student', '/student/time-log', '/student/jobs'], mentor: ['/mentor/mentees'], alumni: ['/alumni'],
  admin: ['/admin', '/admin/students', '/admin/governance'] };
const browser = await chromium.launch({ executablePath: '/opt/pw-browsers/chromium-1194/chrome-linux/chrome' });
const page = await browser.newPage({ viewport: { width: 1366, height: 900 } });
const errors = []; page.on('pageerror', e => errors.push(String(e)));
const api = []; page.on('response', r => { if (r.url().includes('/api/') && r.status() >= 400) api.push(`${r.status()} ${r.request().method()} ${new URL(r.url()).pathname}`); });
let n = 0;
async function step(label) {
  n++; await page.waitForLoadState('networkidle').catch(() => {});
  const f = `${out}/${cid}-${String(n).padStart(2, '0')}-${label}.png`; await page.screenshot({ path: f, fullPage: false });
  const h1 = await page.locator('h1').first().textContent({ timeout: 3000 }).catch(() => null);
  const alerts = await page.getByRole('alert').allTextContents().catch(() => []);
  console.log(`step ${n} ${label}: url=${page.url().replace(BASE, '')} h1=${JSON.stringify(h1 && h1.trim())} alerts=${JSON.stringify(alerts.map(a => a.trim()).filter(Boolean))} screenshot=${f.split('/').pop()}`);
}
const [email, pw, portal] = A[role];
await page.goto(BASE + '/login'); await step('login-page');
if (portal) await page.getByRole('radio', { name: portal }).click();
else await page.getByRole('button', { name: /Main Admin/ }).click();
await page.locator('input[type=email], input[autocomplete=username], input[type=text]').first().fill(email);
await page.getByPlaceholder('Enter your password').fill(pw);
await step('form-filled');
await page.getByRole('button', { name: 'Sign in', exact: true }).click();
await page.waitForURL(u => !u.pathname.startsWith('/login'), { timeout: 15000 });
await step('signed-in');
for (const s of SCREENS[role]) { await page.goto(BASE + s); await step(s.replaceAll('/', '_').slice(1)); }
if (process.env.EXTRA) { const m = await import(process.env.EXTRA); await m.default(page, step); }
const names = await page.getByRole('banner').getByRole('button').allTextContents(); console.log('banner buttons:', JSON.stringify(names.map(s => s.trim())));
const direct = page.getByRole('banner').getByRole('button', { name: /Sign out/ });
if (await direct.count()) await direct.first().click();
else { await page.getByRole('banner').getByRole('button', { name: ACCT[role] }).click(); await page.getByRole('menuitem', { name: 'Sign out' }).click(); }
await page.waitForURL(u => u.pathname.startsWith('/login'), { timeout: 15000 }).catch(() => {});
await step('signed-out');
await page.goto(BASE + SCREENS[role][0]); await step('protected-after-signout');
console.log('page errors:', JSON.stringify(errors)); console.log('API responses >= 400:', JSON.stringify(api));
await browser.close();
