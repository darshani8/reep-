# Voice worker (voice_agent.py) — install in a SEPARATE environment from the
# FastAPI server. The audio/ML stack is heavy and wants Python 3.12:
#
#   py -3.12 -m venv .venv-voice
#   .venv-voice/Scripts/pip install -r requirements-voice.txt
#
# ARCHITECTURE: a four-stage CASCADE, not native speech-to-speech.
#   LiveKit BVC noise cancellation -> Silero VAD -> Groq Whisper (STT)
#   -> Groq Llama (LLM) -> Edge TTS or Groq TTS
# Gemini Live was the original design; this Google project is denied access to
# it (WebSocket 1008 "project has been denied access"), so the Gemini plugin is
# NOT used and is deliberately not installed.
#
# EVERY import in voice_agent.py must appear below. The file previously declared
# only livekit-agents[google] while importing groq, silero, noise_cancellation
# and edge_tts, so a clean checkout installed a worker that raised ImportError on
# startup — it only ran here because those packages had been pip-installed by
# hand into the local venv. Pinned exactly (==) rather than ~= : the SDK contract
# below was verified against these versions by introspecting the installed
# package, and ~=1.6 silently admits the whole 1.x series (that is how the
# earlier ~=1.5 pin let 1.6.10 in unnoticed).
#
# VERIFIED against livekit-agents 1.6.10 — what voice_agent.py relies on:
#   AgentServer().rtc_session(agent_name=...)              exists
#   agents.cli.run_app(server)                             exists
#   AgentSession(vad=, stt=, llm=, tts=, turn_handling=)   all real kwargs
#   AgentSession.start(agent=, room=, room_options=)       RoomInputOptions is
#                                                          deprecated; use
#                                                          RoomOptions(audio_input=...)
#   AgentSession.say(text, allow_interruptions=, add_to_chat_ctx=)
#   session.on("conversation_item_added") -> ConversationItemAddedEvent
#   event.item is a ChatMessage: .id, .role, .content (a LIST), .text_content
#     (property -> str | None). No is_final/interim flag — the event fires only
#     for committed items, so _extract_turn assuming final is correct here.
#   tts.TTS / tts.ChunkedStream / AudioEmitter.initialize(...)  the contract the
#     EdgeTTS adapter implements.
#   Adaptive interruption requires stt.capabilities.streaming AND
#     aligned_transcript; Groq Whisper has NEITHER, so interruption mode must be
#     "vad" or the SDK silently disables the detector.
# On any version bump, re-run that introspection BEFORE the first live call.

livekit-agents==1.6.10

# STT (whisper-large-v3-turbo) and LLM (llama-3.3-70b-versatile), plus the Groq
# TTS used when VOICE_TTS=groq. One GROQ_API_KEY drives all three.
livekit-plugins-groq==1.6.10

# Turn detection. Runs locally; no key, no network.
livekit-plugins-silero==1.6.10

# Server-side background voice cancellation (BVC). Strips the agent's own voice
# echoing back through laptop speakers, which otherwise reads as the student
# barging in and cuts the agent off mid-sentence.
livekit-plugins-noise-cancellation==0.3.0

# Default TTS. NOTE: edge-tts is an UNOFFICIAL, no-SLA endpoint with no privacy
# terms or quota guarantees — acceptable for development and demos, NOT for a
# student cohort. Production must set VOICE_TTS=groq (or an SLA-backed provider)
# and this dependency should then be dropped.
edge-tts==7.2.8

# No HTTP client is listed: the worker talks to the FastAPI server with stdlib
# urllib only (POST /api/voice/transcript + /api/voice/heartbeat), so it carries
# no DB deps and no extra HTTP dependency.
