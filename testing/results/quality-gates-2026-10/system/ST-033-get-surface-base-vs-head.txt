"""Call every parameter-free GET in the OpenAPI document as each role (and anonymous) against two servers; compare status, JSON shape, X-Reep-* headers."""
import json, sys, requests
SPEC, A, B, OUT = sys.argv[1], sys.argv[2], sys.argv[3], sys.argv[4]
ACC = {'student': ('student@bgscet.ac.in', 'student123'), 'mentor': ('mentor@bgscet.ac.in', 'mentor123'),
       'alumni': ('alumni@bgscet.ac.in', 'alumni123'), 'admin': ('admin@bgscet.ac.in', 'admin123'), 'anonymous': None}
SKIP = {'/api/auth/sso/google', '/api/auth/sso/google/callback'}  # external redirect with a random state
spec = json.load(open(SPEC))
paths = sorted(p for p, v in spec['paths'].items() if 'get' in v and '{' not in p and p not in SKIP
               and not any(prm.get('required') for prm in v['get'].get('parameters', []) if prm.get('in') == 'query'))
def shape(x, depth=0):
    if depth > 4: return '…'
    if isinstance(x, dict): return {k: shape(v, depth + 1) for k, v in sorted(x.items())}
    if isinstance(x, list): return ['list'] if not x else [shape(x[0], depth + 1)]
    return type(x).__name__ if x is not None else 'null'
def probe(base, role):
    s = requests.Session()
    if ACC[role]:
        r = s.post(f'{base}/api/auth/login', json=dict(zip(('email', 'password'), ACC[role])), timeout=30); assert r.status_code == 200, r.text
    res = {}
    for p in paths:
        r = s.get(base + p, timeout=120, allow_redirects=False)
        ct = r.headers.get('content-type', '')
        try: body = shape(r.json()) if 'json' in ct else ct.split(';')[0]
        except Exception: body = 'unparseable'
        hdr = sorted((k.lower(), v if k.lower() == 'x-reep-scope' else '*') for k, v in r.headers.items() if k.lower().startswith('x-reep'))
        res[p] = (r.status_code, body, hdr)
    if ACC[role]: s.post(f'{base}/api/auth/logout')
    return res
lines, diffs, total = [], 0, 0
for role in ACC:
    ra, rb = probe(A, role), probe(B, role)
    for p in paths:
        total += 1
        same = ra[p] == rb[p]
        if not same: diffs += 1
        lines.append(f"{'SAME' if same else 'DIFF'}  {role:9s} GET {p}  base={ra[p][0]} head={rb[p][0]}" + ('' if same else f"\n      base: {json.dumps(ra[p])[:600]}\n      head: {json.dumps(rb[p])[:600]}"))
codes = {}
for l in lines:
    pass
open(OUT, 'w').write('\n'.join(lines) + '\n')
print(f'paths={len(paths)} roles={len(ACC)} comparisons={total} differences={diffs}')
