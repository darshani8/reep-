import sys, xml.etree.ElementTree as ET, collections
def load(p):
    out = {}
    for tc in ET.parse(p).iter('testcase'):
        k = tc.get('classname') + '::' + tc.get('name')
        st = 'failed' if tc.find('failure') is not None or tc.find('error') is not None else 'skipped' if tc.find('skipped') is not None else 'passed'
        out[k] = st
    return out
b, h = load(sys.argv[1]), load(sys.argv[2])
print('base', len(b), dict(collections.Counter(b.values())))
print('head', len(h), dict(collections.Counter(h.values())))
lost = [k for k, v in b.items() if v == 'passed' and h.get(k) != 'passed']
print('passed on base but not passed on head:', len(lost)); [print('  ', k, '->', h.get(k, 'ABSENT')) for k in lost]
new = sorted(set(h) - set(b)); print('new on head:', len(new))
mods = collections.Counter(k.split('::')[0] for k in new); [print(f'   {n:4d} {m}') for m, n in mods.most_common()]
gone = sorted(set(b) - set(h)); print('on base but absent on head:', len(gone)); [print('  ', k) for k in gone]
