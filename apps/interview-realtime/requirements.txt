# REEP — real-time mock-interview relay. RUNTIME dependencies only.
#
# Pinned exactly (==), matching apps/api-py/requirements.txt. A `>=` bound means
# a rebuild months from now resolves a dependency set nobody has run against a
# live OpenAI Realtime session, and the first sign of the drift is a production
# call that connects, transcribes, and plays no sound.
#
# Bumping: change the pin, run a real end-to-end call (see README "Smoke test"),
# commit both together.
#
# Python 3.11 is the floor: the relay uses asyncio.TaskGroup, `except*` exception
# groups and asyncio.timeout, all of which are 3.11 language/stdlib features.

fastapi==0.141.1

# ASGI server. The [standard] extra brings httptools + uvloop (POSIX) and the
# server side of the browser WebSocket. Same pin as apps/api-py so the two
# services cannot disagree about WebSocket behaviour.
uvicorn[standard]==0.52.3

# Upstream client for wss://api.openai.com/v1/realtime.
#
# THE VERSION TRAP: websockets >= 14.0 RENAMED the connect() keyword
# `extra_headers=` to `additional_headers=`. On 15.0.1 the old name raises
# TypeError at CONNECT time, not import time — i.e. it survives CI and fails on
# the first student's first call. app/server.py uses `additional_headers=`, so
# THIS PIN AND THAT KEYWORD MOVE TOGETHER. Downgrading below 14 without renaming
# the keyword breaks every interview.
#
# 15.0.1 is also the release line with the modern asyncio API
# (websockets.asyncio.client.connect / ClientConnection) that the relay imports.
#
# uvicorn[standard] depends on `websockets` for the SERVER side of the browser
# socket as well, so this explicit pin governs both directions. After any bump,
# start uvicorn and confirm it logs no `--ws` implementation warning.
websockets==15.0.1

# Pinned explicitly even though fastapi and pydantic-settings both pull it in.
# app/config.py is a pydantic v2 model and uses v2-only API (field_validator with
# mode="before", ValidationInfo, model_fields), so a resolver free to pick v1
# would fail at import with an error far from the cause.
#
# NOTE: an earlier version of this comment justified the pin by "the event models
# in app/types.py". There is no app/types.py — app/ contains config.py and
# server.py, and the Realtime events are handled as plain dicts against the
# allowlists at the top of server.py. The pin is right; the reason was not.
pydantic==2.13.4

# Settings: SCREAMING_SNAKE env vars -> snake_case fields, case-insensitively,
# with .env support. Same convention as apps/api-py/app/config.py.
pydantic-settings==2.15.0

# .env file parsing. pydantic-settings depends on it transitively; pinned here
# because `SettingsConfigDict(env_file=...)` silently does NOTHING if it is
# absent — the app would start with every setting at its blank default and the
# only symptom would be a 4001 close with a .env sitting right there.
# Same version already resolved into the apps/api-py venv, so the two services
# parse .env files identically.
python-dotenv==1.2.2
