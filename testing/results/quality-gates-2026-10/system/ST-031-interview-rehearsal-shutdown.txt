"""Open the interview socket as ROLE and report what happens. Usage: ws_probe.py BASE ROLE SECONDS [specialization]"""
import asyncio, sys, time, requests, websockets
BASE, ROLE, SECS = sys.argv[1], sys.argv[2], float(sys.argv[3]); SPEC = sys.argv[4] if len(sys.argv) > 4 else 'hr'
ACC = {'student': ('student@bgscet.ac.in', 'student123'), 'admin': ('admin@bgscet.ac.in', 'admin123'), 'mentor': ('mentor@bgscet.ac.in', 'mentor123')}
s = requests.Session(); r = s.post(f'{BASE}/api/auth/login', json=dict(zip(('email', 'password'), ACC[ROLE]))); print('login', r.status_code)
print('status', s.get(f'{BASE}/api/interview/status').json())
cookie = '; '.join(f'{c.name}={c.value}' for c in s.cookies)
async def main():
    url = BASE.replace('http', 'ws') + f'/api/interview?specialization={SPEC}'
    t0 = time.time()
    try:
        async with websockets.connect(url, additional_headers={'Cookie': cookie, 'Origin': 'http://localhost:4200'}, max_size=None) as ws:
            print(f'{time.time()-t0:5.2f}s connected')
            while time.time() - t0 < SECS:
                try: m = await asyncio.wait_for(ws.recv(), timeout=SECS)
                except asyncio.TimeoutError: print(f'{time.time()-t0:5.2f}s still open, nothing received'); break
                print(f'{time.time()-t0:5.2f}s recv', (m[:200] if isinstance(m, str) else f'<{len(m)} bytes>'))
    except websockets.ConnectionClosed as e:
        print(f'{time.time()-t0:5.2f}s closed code={e.rcvd.code if e.rcvd else None} reason={e.rcvd.reason if e.rcvd else None!r}')
    except Exception as e:
        print(f'{time.time()-t0:5.2f}s error {type(e).__name__}: {e}')
asyncio.run(main())
